cmake_minimum_required(VERSION 3.24)
project(WispNativeTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Gui Quick Qml Test WebEngineQuick)
enable_testing()
get_filename_component(WISP_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
foreach(name editor settings)
    add_executable(${name}_test ${name}_test.cpp)
    target_include_directories(${name}_test PRIVATE "${WISP_SOURCE_DIR}/native/include")
    target_compile_definitions(${name}_test PRIVATE WISP_SOURCE_DIR="${WISP_SOURCE_DIR}")
    if(MSVC)
        target_compile_options(${name}_test PRIVATE /utf-8)
    endif()
    target_link_libraries(${name}_test PRIVATE Qt6::Gui Qt6::Quick Qt6::Qml Qt6::Test)
    # Qt Test otherwise sends its text output to the debugger on Windows.
    if(WIN32)
        # GUI-subsystem programs may have no stdout handle under CTest.
        add_test(NAME ${name} COMMAND ${name}_test -o ${name}-results.txt,txt)
    else()
        add_test(NAME ${name} COMMAND ${name}_test -o -,txt)
    endif()
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 30)
endforeach()
target_link_libraries(editor_test PRIVATE Qt6::WebEngineQuick)
